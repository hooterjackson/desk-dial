cmake_minimum_required(VERSION 3.20)
project(nanod_lcd_preview LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
get_filename_component(FIRMWARE "${CMAKE_CURRENT_SOURCE_DIR}/../firmware" ABSOLUTE)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
# Firmware sources compiled into the harness. Always the real src/ for evidence
# (build.py passes -DNANOD_FIRMWARE_SRC and -DNANOD_TJPGD_SHIM explicitly, so an
# earlier override cached in build/ cannot survive); overridable (in a separate
# build directory) only to try the renderer against a source overlay.
set(NANOD_FIRMWARE_SRC "${FIRMWARE}/src" CACHE PATH "Firmware src/ directory compiled into the harness")
set(LVGL "${FIRMWARE}/.pio/libdeps/nanofoc_d/lvgl")
set(ARDUINOJSON "${FIRMWARE}/.pio/libdeps/nanofoc_d/ArduinoJson/src")
set(COMPANION "${WORKSPACE}/app")
set(FIXTURES "${COMPANION}/assets/fixtures")
set(FRAMES "${COMPANION}/tests/fixtures/cc5_frames.json")
# cc5.4 (PRESENTATION_V5.md section 15.3): the v5 fixture and the knob copy table.
set(FRAMES_V5 "${COMPANION}/tests/fixtures/frames_v5.json")
set(COPY_TABLE "${CMAKE_CURRENT_SOURCE_DIR}/cc54_copy.json")
# Presentation 6 (Desk Dial r3 release 1): the parser fixture and the r3 knob screens (r3-handoff).
set(FRAMES_V6 "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/frames_v6.json")
set(R3_SCREENS "${CMAKE_CURRENT_SOURCE_DIR}/r3-handoff/r3_screens.json")
# artwork2 fixtures (240 px JPEG covers, 32x32 RGB565 icons; make_art_fixtures.py --artwork2).
set(ARTWORK2_FIXTURES "${CMAKE_CURRENT_SOURCE_DIR}/artwork2")

# LVGL 9.0.0 exactly as the firmware pins it. lv_conf.h next to this file is
# generated by build.py from the firmware's include/lv_conf.h (see README.md).
# Third-party headers are SYSTEM (MSVC /external:I) so /W4 applies to our code only.
file(GLOB_RECURSE LVGL_SOURCES "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
# NANOD_LV_CONF_DIR: where lv_conf.h comes from (this directory: build.py's generated copy of the firmware
# file). perf.py --conf points a separate build directory at an experimental copy (a candidate lv_conf.h change
# measured before it is proposed for the firmware); evidence builds never set it.
set(NANOD_LV_CONF_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Directory holding the lv_conf.h the harness compiles")
target_include_directories(lvgl SYSTEM PUBLIC "${LVGL}" "${NANOD_LV_CONF_DIR}")
if(NOT NANOD_LV_CONF_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  # Every unit must see the same file (MSVC finds a quoted "lv_conf.h" next to an includer first).
  target_compile_definitions(lvgl PUBLIC "LV_CONF_PATH=${NANOD_LV_CONF_DIR}/lv_conf.h")
endif()
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(lvgl PRIVATE /W0)
endif()

# Shared firmware units, compiled unchanged. MSVC cannot go below C++14; the
# device's gnu++11 is enforced separately by cpp11_gate.py (run by build.py).
# An OBJECT library so every unit is linked, even before anything calls it.
add_library(firmware_shared OBJECT
  "${NANOD_FIRMWARE_SRC}/cc_display.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_icons.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_icon_morph.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_crumbs.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_frame_parse.cpp")
set_target_properties(firmware_shared PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(firmware_shared PUBLIC "${NANOD_FIRMWARE_SRC}")
target_include_directories(firmware_shared SYSTEM PUBLIC "${ARDUINOJSON}")
target_link_libraries(firmware_shared PUBLIC lvgl)
if(MSVC)
  # ArduinoJson's std::ostream support needs C++ unwinding (C4530 otherwise).
  target_compile_options(firmware_shared PRIVATE /EHsc)
  # The renderer and icon tables must stay /W4 /WX clean (contract section 10).
  set_source_files_properties("${NANOD_FIRMWARE_SRC}/cc_display.cpp" "${NANOD_FIRMWARE_SRC}/cc_icons.cpp"
    PROPERTIES COMPILE_OPTIONS "/W4;/WX")
endif()
# Generated knob fonts (src/fonts/cc_font_*.c, hybrid: ASCII via the built-ins).
file(GLOB CC_FONTS "${NANOD_FIRMWARE_SRC}/fonts/cc_font_*.c")
if(NOT CC_FONTS)
  message(FATAL_ERROR "No generated fonts under ${NANOD_FIRMWARE_SRC}/fonts (run gen_lvgl_font.py)")
endif()
target_sources(firmware_shared PRIVATE ${CC_FONTS})

# The v5_bounded reference (PRESENTATION_V5.md 8.2 R2, 15.3): the same units with every animated
# layer 240 x 240 (CC_DISPLAY_FULL_LAYERS=1); cc54_report.py compares its renders pixel for pixel.
add_library(firmware_shared_full OBJECT
  "${NANOD_FIRMWARE_SRC}/cc_display.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_icons.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_icon_morph.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_crumbs.cpp"
  "${NANOD_FIRMWARE_SRC}/cc_frame_parse.cpp" ${CC_FONTS})
set_target_properties(firmware_shared_full PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(firmware_shared_full PUBLIC "${NANOD_FIRMWARE_SRC}")
target_include_directories(firmware_shared_full SYSTEM PUBLIC "${ARDUINOJSON}")
target_compile_definitions(firmware_shared_full PUBLIC CC_DISPLAY_FULL_LAYERS=1)
target_link_libraries(firmware_shared_full PUBLIC lvgl)
if(MSVC)
  target_compile_options(firmware_shared_full PRIVATE /EHsc)
endif()

# artwork2 JPEG glue (ARTWORK2.md sections 7 and 10): the firmware's own
# src/cc_jpeg.cpp, compiled against tjpgd_shim/ (fw-core), which maps the ESP ROM
# esp_rom_tjpgd_* API onto LVGL's bundled TJpgDec (src/libs/tjpgd, already in the
# lvgl library above). Every .c/.cpp in tjpgd_shim/ is compiled; its headers
# shadow the ESP ones. Until both exist the harness decodes with its own
# stand-in (jpeg_standin.cpp, the same TJpgDec) and records NANOD_JPEG_GLUE
# "standin", which cc5_report.py fails (jpeg_glue). NANOD_TJPGD_SHIM, like
# NANOD_FIRMWARE_SRC, is overridable only to try a shim change in a separate
# build directory.
set(NANOD_TJPGD_SHIM "${CMAKE_CURRENT_SOURCE_DIR}/tjpgd_shim" CACHE PATH "Host shim of the ESP ROM TJpgDec API")
set(TJPGD_SHIM "${NANOD_TJPGD_SHIM}")
if(EXISTS "${NANOD_FIRMWARE_SRC}/cc_jpeg.cpp" AND IS_DIRECTORY "${TJPGD_SHIM}")
  file(GLOB TJPGD_SHIM_SOURCES "${TJPGD_SHIM}/*.c" "${TJPGD_SHIM}/*.cpp")
  add_library(jpeg_glue OBJECT "${NANOD_FIRMWARE_SRC}/cc_jpeg.cpp" ${TJPGD_SHIM_SOURCES})
  target_include_directories(jpeg_glue BEFORE PUBLIC "${TJPGD_SHIM}")
  target_include_directories(jpeg_glue PUBLIC "${NANOD_FIRMWARE_SRC}")
  target_include_directories(jpeg_glue SYSTEM PRIVATE "${LVGL}/src/libs/tjpgd")
  set(NANOD_JPEG_GLUE "cc_jpeg.cpp")
  set(NANOD_JPEG_FIRMWARE 1)
else()
  message(WARNING "lcd-preview: ${NANOD_FIRMWARE_SRC}/cc_jpeg.cpp or ${TJPGD_SHIM} missing; "
                  "decoding artwork2 covers with the harness stand-in (cc5_report.py jpeg_glue fails)")
  add_library(jpeg_glue OBJECT jpeg_standin.cpp)
  set(NANOD_JPEG_GLUE "standin")
  set(NANOD_JPEG_FIRMWARE 0)
endif()
set_target_properties(jpeg_glue PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(jpeg_glue PUBLIC lvgl)

# Harness driver only: C++17 for std::filesystem. The cc4-era old_display.cpp
# reference renders were dropped in Stage 6 (the file stays for reference).
add_executable(lcd-preview main.cpp)
set_target_properties(lcd-preview PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(lcd-preview PRIVATE firmware_shared jpeg_glue lvgl)
add_executable(lcd-preview-full main.cpp)
set_target_properties(lcd-preview-full PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(lcd-preview-full PRIVATE firmware_shared_full jpeg_glue lvgl)
# NANOD_FIRMWARE_SRC_DIR / NANOD_TJPGD_SHIM_DIR: the directories actually compiled,
# recorded in render-index.json (cc5_report.py jpeg_glue rejects any other ones).
foreach(target lcd-preview lcd-preview-full)
  target_compile_definitions(${target} PRIVATE NANOD_FIXTURE_DIR="${FIXTURES}" NANOD_FRAMES_JSON="${FRAMES}"
    NANOD_FRAMES_V5_JSON="${FRAMES_V5}" NANOD_COPY_JSON="${COPY_TABLE}"
    NANOD_FRAMES_V6_JSON="${FRAMES_V6}" NANOD_R3_SCREENS_JSON="${R3_SCREENS}"
    NANOD_ARTWORK2_DIR="${ARTWORK2_FIXTURES}" NANOD_JPEG_GLUE="${NANOD_JPEG_GLUE}"
    NANOD_JPEG_FIRMWARE=${NANOD_JPEG_FIRMWARE}
    NANOD_FIRMWARE_SRC_DIR="${NANOD_FIRMWARE_SRC}" NANOD_TJPGD_SHIM_DIR="${TJPGD_SHIM}")
  if(MSVC)
    # The cached CMAKE_CXX_FLAGS in build/ is empty, so request C++ unwinding explicitly.
    target_compile_options(${target} PRIVATE /EHsc /utf-8)
  endif()
endforeach()
target_compile_definitions(lcd-preview-full PRIVATE CC_DISPLAY_FULL_LAYERS=1)

# r4 render-speed profile (perf.cpp, run by perf.py): the same firmware units, no fixtures but the v1 covers.
add_executable(lcd-perf perf.cpp)
set_target_properties(lcd-perf PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(lcd-perf PRIVATE firmware_shared lvgl)
target_compile_definitions(lcd-perf PRIVATE NANOD_FIXTURE_DIR="${FIXTURES}")
if(MSVC)
  target_compile_options(lcd-perf PRIVATE /EHsc /utf-8)
endif()

# Optimisation pinned on the profiled code (HN-PERF-001). build/'s CMakeCache was configured long ago with empty
# CMAKE_<LANG>_FLAGS_RELEASE, so its "Release" compiled at MSVC's default /Od while a fresh build-perf-conf got
# CMake's /O2 /Ob2 /DNDEBUG: a compiler gain was credited to an lv_conf.h change. Pinned per target, an empty
# cache can no longer strip them; perf.py records the flags each target was built with and refuses /Od.
if(MSVC)
  foreach(target lvgl firmware_shared lcd-perf)
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:/O2;/Ob2>")
    target_compile_definitions(${target} PRIVATE "$<$<CONFIG:Release>:NDEBUG>")
  endforeach()
  # build/'s empty cache also links Release incrementally (jump thunks); a fresh cache does not.
  target_link_options(lcd-perf PRIVATE "$<$<CONFIG:Release>:/INCREMENTAL:NO>")
endif()
